        function parseTimeToMinutes(timeStr) {

            if (!timeStr) return 0;

            const parts = timeStr.split(":");

            return parseInt(parts[0]) * 60 + parseInt(parts[1]);

        }


        function formatMinutesToClock(minutes) {

            let normalized = (minutes + 1440) % 1440;

            const h = Math.floor(normalized / 60).toString().padStart(2, "0");

            const m = Math.floor(normalized % 60).toString().padStart(2, "0");

            return `${h}:${m}`;

        }

        // CORE WORKOUT LOG PACKAGING LOGIC

        function executeWorkoutEntrySubmission() {

            if (!state.workoutSelectedExercise) {

                alert("Please select an active target movement sequence from the Exercise list panel first.");

                return;

            }


            // Collect structured field parameters

            const s1W = parseFloat(document.getElementById("set1Weight").value);

            const s1R = parseInt(document.getElementById("set1Reps").value);

            const s2W = parseFloat(document.getElementById("set2Weight").value);

            const s2R = parseInt(document.getElementById("set2Reps").value);

            const s3W = parseFloat(document.getElementById("set3Weight").value);

            const s3R = parseInt(document.getElementById("set3Reps").value);

            const notes = document.getElementById("workoutNotes").value;


            if (!s1W || !s1R || !s2W || !s2R || !s3W || !s3R) {

                alert("Please provide valid non-zero inputs for all three working training data arrays.");

                return;

            }


            // Calculate individualized session 1RM metrics using the Epley Equation [w * (1 + r / 30)]

            const max1 = s1W * (1 + s1R / 30);

            const max2 = s2W * (1 + s2R / 30);

            const max3 = s3W * (1 + s3R / 30);

            

            // Peak efficiency tracking value validation anchor

            const bestSessionMax = Math.max(max1, max2, max3);


            // Determine strategy intensity category tag for database historical entries

            let statusText = `Wk ${state.workoutActiveWeek}`;

            let themeContext = "Yellow";

            

            let blockMod = state.workoutActiveWeek % 4;

            if (blockMod === 0) blockMod = 4;

            if (blockMod === 2) themeContext = "Orange";

            if (blockMod === 3) themeContext = "Green";

            if (blockMod === 4) themeContext = "Red";


            if (state.workoutFatigueMode) {

                statusText = "Fatigue Session";

                themeContext = "Grey";

            }

            if (state.workoutTestWeekMode) {

                statusText = "Test Week";

                themeContext = "Purple";

            }


            // Package final operational JSON document item

            const workoutLogPacket = {

                id: "wk_" + new Date().getTime() + "_" + Math.floor(Math.random() * 100),

                routine: state.workoutActiveRoutine,

                week: state.workoutActiveWeek,

                statusLabel: statusText,

                theme: themeContext,

                exercise: state.workoutSelectedExercise,

                set1: { kg: s1W, reps: s1R },

                set2: { kg: s2W, reps: s2R },

                set3: { kg: s3W, reps: s3R },

                calculated1RM: bestSessionMax,

                notes: notes,

                timestamp: new Date().toLocaleDateString()

            };


            // Commit packet to historical array indices unshifted

            state.workoutRegistry.unshift(workoutLogPacket);

            saveState();


            // Refresh presentation layers immediately

            renderWorkoutRegistryTable();

            renderWorkoutRoutineUI();

            calculateActiveWorkoutTension();


            // Clean down inputs form panel

            document.getElementById("workoutNotes").value = "";

            document.getElementById("set1Reps").value = "";

            document.getElementById("set2Reps").value = "";

            document.getElementById("set3Reps").value = "";

        }
